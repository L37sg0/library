## Finishing the RSS feed
- change `feed.py` to look like this:
  ```python
  import yaml
  import xml.etree.ElementTree as xml_tree
  
  with open('feed.yaml', 'r') as file:
      yaml_data = yaml.safe_load(file)
  
      rss_element = xml_tree.Element('rss', {
          'version':'2.0',
          'xmlns:itunes':'https://www.itunes.com/dtds/podcast-1.0.dtg',
          'xmlns:content':'https://purl.org/rss/1.0/modules/content'
      })
  
      channel_element = xml_tree.SubElement(rss_element, 'channel')
  
      link_prefix = yaml_data['link']
      for k in yaml_data:
          element_attributes = {}
          if k == 'item':
              for x in yaml_data[k]:
                  item_element = xml_tree.SubElement(channel_element, 'item')
                  item_data = x
                  for y in item_data:
                      element_attributes = {}
                      if y == 'file':
                          enclosure_element = xml_tree.SubElement(item_element, 'enclosure',{
                              'url': link_prefix + item_data['file'],
                              'type': yaml_data['format'],
                              'length': item_data['length']
                          })
                          continue
                      elif y == 'length':
                          continue
                      elif y == 'duration':
                          element_title = 'itunes:duration'
                      elif y == 'published':
                          element_title = 'pubDate'
                      else:
                          element_title = y
                      xml_tree.SubElement(item_element, element_title, element_attributes).text = item_data[y]
                  
                  xml_tree.SubElement(item_element, 'itunes:author').text = yaml_data['author']
                  
          else:
              if k == 'author':
                  element_title = 'itunes:author'
              elif k == 'image':
                  enclosure_element = xml_tree.SubElement(channel_element, 'itunes:image',{
                      'href': link_prefix + yaml_data['image']
                  })
                  continue
              elif k == 'category':
                  element_title = 'itunes:category'
              else:
                  element_title = k
  
              xml_tree.SubElement(channel_element, element_title, element_attributes).text = yaml_data[k]
  
  output_tree = xml_tree.ElementTree(rss_element)
  output_tree.write('podcast', encoding='UTF-8', xml_declaration=True)
  ```
