## Examples
  ```bash
  $ uname -a     # System info
  Linux codespaces-d2c3e6 6.2.0-1019-azure #19~22.04.1-Ubuntu SMP Wed Jan 10 22:57:03 UTC 2024 x86_64 x86_64 x86_64 GNU/Linux
  ```
  ```bash
  $ uname -r      # Kernel version
  6.2.0-1019-azure
  ```
  ```bash
  $ ls -l /etc/*release    # Place where OS distribution information can be found
  -rw-r--r-- 1 root root 104 Jan  2 17:12 /etc/lsb-release
  lrwxrwxrwx 1 root root  21 Jan  2 17:12 /etc/os-release -> ../usr/lib/os-release
  ```
  ```bash
  $ cat /etc/lsb-release 
  DISTRIB_ID=Ubuntu
  DISTRIB_RELEASE=20.04
  DISTRIB_CODENAME=focal
  DISTRIB_DESCRIPTION="Ubuntu 20.04.6 LTS"
  ```
  ```bash
  $ cat /etc/os-release 
  NAME="Ubuntu"
  VERSION="20.04.6 LTS (Focal Fossa)"
  ID=ubuntu
  ID_LIKE=debian
  PRETTY_NAME="Ubuntu 20.04.6 LTS"
  VERSION_ID="20.04"
  HOME_URL="https://www.ubuntu.com/"
  SUPPORT_URL="https://help.ubuntu.com/"
  BUG_REPORT_URL="https://bugs.launchpad.net/ubuntu/"
  PRIVACY_POLICY_URL="https://www.ubuntu.com/legal/terms-and-policies/privacy-policy"
  VERSION_CODENAME=focal
  UBUNTU_CODENAME=focal
  ```
