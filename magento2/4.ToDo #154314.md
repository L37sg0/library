    - **Notes**:
        - seems the minimum requirement of `system.xml` to show your new section under choosen tab is to have at least one group fully defined
        - example
        ```xml
        <?xml version="1.0" ?>
        <config xmlns:xsi="http://www.w3.org/2001/XMLSchema-instance"
                xsi:noNamespaceSchemaLocation="urn:magento:module:Magento_Config:etc/system_file.xsd">
            <system>
                <tab id="stenik" translate="label" sortOrder="10">
                    <label>Stenik</label>
                </tab>
                <section id="stenik_customer_details" translate="label" sortOrder="131" showInDefault="1" showInWebsite="1" showInStore="1" >
                    <class>separator-top</class>
                    <label>Customer Details</label>
                    <tab>stenik</tab>
                    <resource>Stenik_CustomerDetails::stenik_customerdetails_config</resource>
                    <group id="general" translate="latel" type="text" sortOrder="10" showInDefault="1" showInWebsite="1" showInStore="1"><!--Section started to appear in admin configuration after the group was fully defined-->
                        <label>General Configuration</label>
                        <field id="public_profile_enabled" translate="label" type="select" sortOrder="10" showInDefault="1" showInWebsite="1" showInStore="1">
                            <label>Enable Public Profiles</label>
                            <source_model>Magento\Config\Model\Config\Source\Yesno</source_model>
                            <comment>Choose should public profiles be accessible.</comment>
                        </field>
                    </group>
                </section>
            </system>
        </config>
        ```
        - after definition you need to run `setup:upgrade`
        - magento will generate record for your new config in table `core_config_data` with path `<section id>/<group id>/<field id>` **on the first `Save` of your new config in the admin panel**

        ```xml
            </item>
            <item name="template" xsi:type="string">templates/form/collapsible</item> <!-- template !! -->
        </argument>
        ```

        - **Do not forget to specify settings for data scope**
        ```xml
        <settings>
            <buttons> <!-- if you have buttons you can place them here -->
                <button name="save" class="Magento\Catalog\Block\Adminhtml\Category\Edit\SaveButton"/>
                <button name="delete" class="Magento\Catalog\Block\Adminhtml\Category\Edit\DeleteButton"/>
            </buttons>

            <dataScope>data</dataScope> <!-- remember dataScope and deps !! -->
            <deps>
                <dep>stenik_customer_details_edit_form.stenik_customer_details_edit_form_data_source</dep>
            </deps>
        </settings>
        ```

        - **Do not forget to specify data source**
        ```xml
        <dataSource name="stenik_customer_details_edit_form_data_source">
            <argument name="data" xsi:type="array">
                <item name="js_config" xsi:type="array">
                    <item name="component" xsi:type="string">Magento_Ui/js/form/provider</item> <!-- some JS provider -->
                </item>
            </argument>
            
            <settings>
                <validateUrl path="catalog/category/validate"/>
                <submitUrl path="stenik_customer_details/grid/save"/>
                <!-- here you point your form submission url -->
            </settings>

            <dataProvider class="Stenik\CustomerDetails\Ui\CustomerDetailDataProvider" name="stenik_customer_details_edit_form_data_source"> <!-- your PHP provider !! -->
                <settings> <!-- arguments which go in your PHP provider constructor -->
                    <requestFieldName>id</requestFieldName>
                    <primaryFieldName>id</primaryFieldName>
                </settings>
            </dataProvider>
        </dataSource>
        ```
